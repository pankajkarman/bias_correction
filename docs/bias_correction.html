bias_correction

Bias-correction methods for NumPy/Pandas series and Xarray objects.

The module adjusts scenario/model data against a trusted reference data set. Xarray inputs can be multi-dimensional, such as (time, lat, lon) climate grids, as long as they include the time dimension used for correction. Implemented methods include:

  • basic_quantile: empirical quantile mapping.
  • modified_quantile: modified quantile mapping with median and IQR scaling.
  • gamma_mapping: scaled distribution mapping for non-negative, precipitation-like data.
  • normal_mapping: scaled distribution mapping for approximately normally distributed data.

Installation

pip install bias-correction

Install the latest code directly from GitHub:

pip install git+https://github.com/pankajkarman/bias_correction.git

Usage

Use BiasCorrection for NumPy arrays or Pandas Series:

from bias_correction import BiasCorrection

bc = BiasCorrection(reference, model, scenario)
corrected = bc.correct(method="modified_quantile")

Use XBiasCorrection for Xarray DataArray or Dataset inputs:

from bias_correction import XBiasCorrection

xbc = XBiasCorrection(reference, model, scenario, dim="time")
corrected = xbc.correct(method="gamma_mapping", lower_limit=0.1)

For multi-dimensional Xarray data, pass the name of the time dimension with dim. Corrections are applied along that dimension at each remaining grid point:

reference = xr.DataArray(ref_values, dims=["time", "lat", "lon"])
model = xr.DataArray(model_values, dims=["time", "lat", "lon"])
scenario = xr.DataArray(scenario_values, dims=["time", "lat", "lon"])

xbc = XBiasCorrection(reference, model, scenario, dim="time")
corrected = xbc.correct(method="modified_quantile")

reference is the observed or trusted data, model is the model data over the reference period, and scenario is the model data to correct.

117def quantile_correction(obs_data, mod_data, sce_data, modified=True):
118    """
119    Correct scenario data with empirical quantile mapping.
120
121    Parameters
122    ----------
123    obs_data : array-like
124        Reference or observed data.
125    mod_data : array-like
126        Model data over the reference period.
127    sce_data : array-like
128        Scenario/model data to correct.
129    modified : bool, default True
130        If True, apply the modified quantile correction using the median and
131        interquartile range adjustment. If False, apply basic quantile mapping.
132
133    Returns
134    -------
135    array-like
136        Corrected scenario data with the same shape as `sce_data`.
137
138    Examples
139    --------
140    Basic quantile mapping:
141
142    ```python
143    corrected = quantile_correction(reference, model, scenario, modified=False)
144    ```
145
146    Modified quantile mapping:
147
148    ```python
149    corrected = quantile_correction(reference, model, scenario, modified=True)
150    ```
151    """
152    cdf = ECDF(mod_data)
153    p = cdf(sce_data) * 100
154    cor = np.subtract(*[np.nanpercentile(x, p) for x in [obs_data, mod_data]])
155    if modified:
156        mid = np.subtract(*[np.nanpercentile(x, 50) for x in [obs_data, mod_data]])
157        g = np.true_divide(*[np.nanpercentile(x, 50) for x in [obs_data, mod_data]])
158
159        iqr_obs_data = np.subtract(*np.nanpercentile(obs_data, [75, 25]))
160        iqr_mod_data = np.subtract(*np.nanpercentile(mod_data, [75, 25]))
161
162        f = np.true_divide(iqr_obs_data, iqr_mod_data)
163        cor = g * mid + f * (cor - mid)
164        return sce_data + cor
165    else:
166        return sce_data + cor

169def gamma_correction(
170    obs_data, mod_data, sce_data, lower_limit=0.1, cdf_threshold=0.9999999
171):
172    """
173    Correct precipitation-like scenario data with gamma distribution mapping.
174
175    Values below `lower_limit` are treated as dry or zero events. Corrected
176    wet-day values are assigned back to the largest scenario values so dry-day
177    frequency is preserved according to the model/reference frequency ratio.
178
179    Parameters
180    ----------
181    obs_data : array-like
182        Reference or observed data.
183    mod_data : array-like
184        Model data over the reference period.
185    sce_data : array-like
186        Scenario/model data to correct.
187    lower_limit : float, default 0.1
188        Minimum value considered a wet day.
189    cdf_threshold : float, default 0.9999999
190        Upper CDF bound used to avoid infinite distribution tails.
191
192    Returns
193    -------
194    numpy.ndarray
195        Corrected scenario data with the same length as `sce_data`.
196
197    Examples
198    --------
199    ```python
200    corrected = gamma_correction(
201        reference,
202        model,
203        scenario,
204        lower_limit=0.1,
205    )
206    ```
207    """
208    obs_raindays, mod_raindays, sce_raindays = [
209        x[x >= lower_limit] for x in [obs_data, mod_data, sce_data]
210    ]
211    obs_gamma, mod_gamma, sce_gamma = [
212        gamma.fit(x) for x in [obs_raindays, mod_raindays, sce_raindays]
213    ]
214
215    obs_cdf = gamma.cdf(np.sort(obs_raindays), *obs_gamma)
216    mod_cdf = gamma.cdf(np.sort(mod_raindays), *mod_gamma)
217    sce_cdf = gamma.cdf(np.sort(sce_raindays), *sce_gamma)
218
219    obs_cdf[obs_cdf > cdf_threshold] = cdf_threshold
220    mod_cdf[mod_cdf > cdf_threshold] = cdf_threshold
221    sce_cdf[sce_cdf > cdf_threshold] = cdf_threshold
222
223    obs_cdf_intpol = np.interp(
224        np.linspace(1, len(obs_raindays), len(sce_raindays)),
225        np.linspace(1, len(obs_raindays), len(obs_raindays)),
226        obs_cdf,
227    )
228
229    mod_cdf_intpol = np.interp(
230        np.linspace(1, len(mod_raindays), len(sce_raindays)),
231        np.linspace(1, len(mod_raindays), len(mod_raindays)),
232        mod_cdf,
233    )
234
235    obs_inverse, mod_inverse, sce_inverse = [
236        1.0 / (1.0 - x) for x in [obs_cdf_intpol, mod_cdf_intpol, sce_cdf]
237    ]
238
239    adapted_cdf = 1 - 1.0 / (obs_inverse * sce_inverse / mod_inverse)
240    adapted_cdf[adapted_cdf < 0.0] = 0.0
241
242    initial = (
243        gamma.ppf(np.sort(adapted_cdf), *obs_gamma)
244        * gamma.ppf(sce_cdf, *sce_gamma)
245        / gamma.ppf(sce_cdf, *mod_gamma)
246    )
247
248    mod_frequency = 1.0 * mod_raindays.shape[0] / mod_data.shape[0]
249    sce_frequency = 1.0 * sce_raindays.shape[0] / sce_data.shape[0]
250
251    days_min = len(sce_raindays) * sce_frequency / mod_frequency
252
253    expected_sce_raindays = int(min(days_min, len(sce_data)))
254
255    sce_argsort = np.argsort(sce_data)
256    correction = np.zeros(len(sce_data))
257
258    if len(sce_raindays) > expected_sce_raindays:
259        initial = np.interp(
260            np.linspace(1, len(sce_raindays), expected_sce_raindays),
261            np.linspace(1, len(sce_raindays), len(sce_raindays)),
262            initial,
263        )
264    else:
265        initial = np.hstack(
266            (np.zeros(expected_sce_raindays - len(sce_raindays)), initial)
267        )
268
269    if expected_sce_raindays > 0:
270        correction[sce_argsort[-expected_sce_raindays:]] = initial
271    # correction = pd.Series(correction, index=sce_data.index)
272    return correction

275def normal_correction(obs_data, mod_data, sce_data, cdf_threshold=0.9999999):
276    """
277    Correct scenario data with normal scaled distribution mapping.
278
279    Each input is detrended before fitting a normal distribution. The corrected
280    values are then shifted so the scenario trend is added back.
281
282    Parameters
283    ----------
284    obs_data : array-like
285        Reference or observed data.
286    mod_data : array-like
287        Model data over the reference period.
288    sce_data : array-like
289        Scenario/model data to correct.
290    cdf_threshold : float, default 0.9999999
291        Lower and upper CDF clipping threshold used to avoid infinite tails.
292
293    Returns
294    -------
295    numpy.ndarray
296        Corrected scenario data with the same length as `sce_data`.
297
298    Examples
299    --------
300    ```python
301    corrected = normal_correction(reference, model, scenario)
302    ```
303    """
304    obs_len, mod_len, sce_len = [len(x) for x in [obs_data, mod_data, sce_data]]
305    obs_mean, mod_mean, sce_mean = [x.mean() for x in [obs_data, mod_data, sce_data]]
306    obs_detrended, mod_detrended, sce_detrended = [
307        detrend(x) for x in [obs_data, mod_data, sce_data]
308    ]
309    obs_norm, mod_norm, sce_norm = [
310        norm.fit(x) for x in [obs_detrended, mod_detrended, sce_detrended]
311    ]
312
313    obs_cdf = norm.cdf(np.sort(obs_detrended), *obs_norm)
314    mod_cdf = norm.cdf(np.sort(mod_detrended), *mod_norm)
315    sce_cdf = norm.cdf(np.sort(sce_detrended), *sce_norm)
316
317    obs_cdf = np.maximum(np.minimum(obs_cdf, cdf_threshold), 1 - cdf_threshold)
318    mod_cdf = np.maximum(np.minimum(mod_cdf, cdf_threshold), 1 - cdf_threshold)
319    sce_cdf = np.maximum(np.minimum(sce_cdf, cdf_threshold), 1 - cdf_threshold)
320
321    sce_diff = sce_data - sce_detrended
322    sce_argsort = np.argsort(sce_detrended)
323
324    obs_cdf_intpol = np.interp(
325        np.linspace(1, obs_len, sce_len), np.linspace(1, obs_len, obs_len), obs_cdf
326    )
327    mod_cdf_intpol = np.interp(
328        np.linspace(1, mod_len, sce_len), np.linspace(1, mod_len, mod_len), mod_cdf
329    )
330    obs_cdf_shift, mod_cdf_shift, sce_cdf_shift = [
331        (x - 0.5) for x in [obs_cdf_intpol, mod_cdf_intpol, sce_cdf]
332    ]
333
334    obs_inverse, mod_inverse, sce_inverse = [
335        1.0 / (0.5 - np.abs(x)) for x in [obs_cdf_shift, mod_cdf_shift, sce_cdf_shift]
336    ]
337
338    adapted_cdf = np.sign(obs_cdf_shift) * (
339        1.0 - 1.0 / (obs_inverse * sce_inverse / mod_inverse)
340    )
341    adapted_cdf[adapted_cdf < 0] += 1.0
342    adapted_cdf = np.maximum(np.minimum(adapted_cdf, cdf_threshold), 1 - cdf_threshold)
343
344    xvals = norm.ppf(np.sort(adapted_cdf), *obs_norm) + obs_norm[-1] / mod_norm[-1] * (
345        norm.ppf(sce_cdf, *sce_norm) - norm.ppf(sce_cdf, *mod_norm)
346    )
347
348    xvals -= xvals.mean()
349    xvals += obs_mean + (sce_mean - mod_mean)
350
351    correction = np.zeros(sce_len)
352    correction[sce_argsort] = xvals
353    correction += sce_diff - sce_mean
354    # correction = pd.Series(correction, index=sce_data.index)
355    return correction

358class BiasCorrection(object):
359    """
360    Bias-correction wrapper for NumPy arrays and Pandas Series.
361
362    Parameters
363    ----------
364    obs_data : array-like
365        Reference or observed data.
366    mod_data : array-like
367        Model data over the reference period.
368    sce_data : array-like
369        Scenario/model data to correct.
370
371    Notes
372    -----
373    If `sce_data` is a Pandas `Series`, corrected output is returned as a
374    `Series` with the original index preserved.
375    """
376
377    def __init__(self, obs_data, mod_data, sce_data):
378        self.obs_data = obs_data
379        self.mod_data = mod_data
380        self.sce_data = sce_data
381
382    def correct(
383        self, method="modified_quantile", lower_limit=0.1, cdf_threshold=0.9999999
384    ):
385        """
386        Apply a bias-correction method.
387
388        Parameters
389        ----------
390        method : {"modified_quantile", "basic_quantile", "gamma_mapping", "normal_mapping"}
391            Correction method to apply.
392        lower_limit : float, default 0.1
393            Wet-day threshold used by `gamma_mapping`.
394        cdf_threshold : float, default 0.9999999
395            CDF clipping threshold used by distribution mapping methods.
396
397        Returns
398        -------
399        array-like or pandas.Series
400            Corrected scenario data.
401
402        Examples
403        --------
404        ```python
405        corrected = bc.correct(method="basic_quantile")
406        corrected = bc.correct(method="modified_quantile")
407        corrected = bc.correct(method="gamma_mapping", lower_limit=0.1)
408        corrected = bc.correct(method="normal_mapping")
409        ```
410        """
411        if method == "gamma_mapping":
412            corrected = gamma_correction(
413                self.obs_data,
414                self.mod_data,
415                self.sce_data,
416                lower_limit=lower_limit,
417                cdf_threshold=cdf_threshold,
418            )
419        elif method == "normal_mapping":
420            corrected = normal_correction(
421                self.obs_data, self.mod_data, self.sce_data, cdf_threshold=cdf_threshold
422            )
423        elif method == "basic_quantile":
424            corrected = quantile_correction(
425                self.obs_data, self.mod_data, self.sce_data, modified=False
426            )
427
428        elif method == "modified_quantile":
429            corrected = quantile_correction(
430                self.obs_data, self.mod_data, self.sce_data, modified=True
431            )
432
433        else:
434            raise Exception("Specify correct method for bias correction.")
435
436        if isinstance(self.sce_data, pd.Series):
437            corrected = pd.Series(corrected, index=self.sce_data.index)
438
439        self.corrected = corrected
440        return self.corrected

443class XBiasCorrection(object):
444    """
445    Bias-correction wrapper for Xarray DataArray and Dataset objects.
446
447    Inputs may be one-dimensional or multi-dimensional. Each input must include
448    the core correction dimension, usually `time`; correction is applied along
449    that dimension while Xarray preserves the remaining coordinates.
450
451    Parameters
452    ----------
453    obs_data : xarray.DataArray or xarray.Dataset
454        Reference or observed data.
455    mod_data : xarray.DataArray or xarray.Dataset
456        Model data over the reference period.
457    sce_data : xarray.DataArray or xarray.Dataset
458        Scenario/model data to correct.
459    dim : str, default "time"
460        Core dimension over which the correction is applied.
461    """
462
463    def __init__(self, obs_data, mod_data, sce_data, dim="time"):
464        self.obs_data = obs_data
465        self.mod_data = mod_data
466        self.sce_data = sce_data
467        self.dim = dim
468
469    def correct(
470        self,
471        method="modified_quantile",
472        lower_limit=0.1,
473        cdf_threshold=0.9999999,
474        vectorize=True,
475        dask="parallelized",
476        **apply_ufunc_kwargs
477    ):
478        """
479        Apply a bias-correction method with `xarray.apply_ufunc`.
480
481        Parameters
482        ----------
483        method : {"modified_quantile", "basic_quantile", "gamma_mapping", "normal_mapping"}
484            Correction method to apply.
485        lower_limit : float, default 0.1
486            Wet-day threshold used by `gamma_mapping`.
487        cdf_threshold : float, default 0.9999999
488            CDF clipping threshold used by distribution mapping methods.
489        vectorize : bool, default True
490            Passed to `xarray.apply_ufunc`.
491        dask : str, default "parallelized"
492            Dask mode passed to `xarray.apply_ufunc`.
493        **apply_ufunc_kwargs
494            Additional keyword arguments forwarded to `xarray.apply_ufunc`.
495
496        Returns
497        -------
498        xarray.DataArray or xarray.Dataset
499            Corrected scenario data with Xarray coordinates preserved.
500
501        Examples
502        --------
503        ```python
504        corrected = xbc.correct(method="basic_quantile")
505        corrected = xbc.correct(method="modified_quantile")
506        corrected = xbc.correct(method="gamma_mapping", lower_limit=0.1)
507        corrected = xbc.correct(method="normal_mapping")
508        ```
509        """
510        dtype = self._set_dtype()
511        dim = self.dim
512        if method == "gamma_mapping":
513            corrected = xr.apply_ufunc(
514                gamma_correction,
515                self.obs_data,
516                self.mod_data,
517                self.sce_data,
518                vectorize=vectorize,
519                dask=dask,
520                input_core_dims=[[dim], [dim], [dim]],
521                output_core_dims=[[dim]],
522                output_dtypes=[dtype],
523                kwargs={"lower_limit": lower_limit, "cdf_threshold": cdf_threshold},
524                **apply_ufunc_kwargs
525            )
526        elif method == "normal_mapping":
527            corrected = xr.apply_ufunc(
528                normal_correction,
529                self.obs_data,
530                self.mod_data,
531                self.sce_data,
532                vectorize=vectorize,
533                dask=dask,
534                input_core_dims=[[dim], [dim], [dim]],
535                output_core_dims=[[dim]],
536                output_dtypes=[dtype],
537                kwargs={"cdf_threshold": cdf_threshold},
538                **apply_ufunc_kwargs
539            )
540        elif method == "basic_quantile":
541            corrected = xr.apply_ufunc(
542                quantile_correction,
543                self.obs_data,
544                self.mod_data,
545                self.sce_data,
546                vectorize=vectorize,
547                dask=dask,
548                input_core_dims=[[dim], [dim], [dim]],
549                output_core_dims=[[dim]],
550                kwargs={"modified": False},
551                **apply_ufunc_kwargs
552            )
553
554        elif method == "modified_quantile":
555            corrected = xr.apply_ufunc(
556                quantile_correction,
557                self.obs_data,
558                self.mod_data,

171    obs_data, mod_data, sce_data, lower_limit=0.1, cdf_threshold=0.9999999
172):
173    """
174    Correct precipitation-like scenario data with gamma distribution mapping.
175
176    Values below `lower_limit` are treated as dry or zero events. Corrected
177    wet-day values are assigned back to the largest scenario values so dry-day
178    frequency is preserved according to the model/reference frequency ratio.
179
180    Parameters
181    ----------
182    obs_data : array-like
183        Reference or observed data.
184    mod_data : array-like
185        Model data over the reference period.
186    sce_data : array-like
187        Scenario/model data to correct.
188    lower_limit : float, default 0.1
189        Minimum value considered a wet day.
190    cdf_threshold : float, default 0.9999999
191        Upper CDF bound used to avoid infinite distribution tails.
192
193    Returns
194    -------
195    numpy.ndarray
196        Corrected scenario data with the same length as `sce_data`.
197
198    Examples
199    --------
200    ```python
201    corrected = gamma_correction(
202        reference,
203        model,
204        scenario,
205        lower_limit=0.1,
206    )
207    ```
208    """
209    obs_raindays, mod_raindays, sce_raindays = [
210        x[x >= lower_limit] for x in [obs_data, mod_data, sce_data]
211    ]
212    obs_gamma, mod_gamma, sce_gamma = [
213        gamma.fit(x) for x in [obs_raindays, mod_raindays, sce_raindays]
214    ]
215
216    obs_cdf = gamma.cdf(np.sort(obs_raindays), *obs_gamma)
217    mod_cdf = gamma.cdf(np.sort(mod_raindays), *mod_gamma)
218    sce_cdf = gamma.cdf(np.sort(sce_raindays), *sce_gamma)
219
220    obs_cdf[obs_cdf > cdf_threshold] = cdf_threshold
221    mod_cdf[mod_cdf > cdf_threshold] = cdf_threshold
222    sce_cdf[sce_cdf > cdf_threshold] = cdf_threshold
223
224    obs_cdf_intpol = np.interp(
225        np.linspace(1, len(obs_raindays), len(sce_raindays)),
226        np.linspace(1, len(obs_raindays), len(obs_raindays)),
227        obs_cdf,
228    )
229
230    mod_cdf_intpol = np.interp(
231        np.linspace(1, len(mod_raindays), len(sce_raindays)),
232        np.linspace(1, len(mod_raindays), len(mod_raindays)),
233        mod_cdf,
234    )
235
236    obs_inverse, mod_inverse, sce_inverse = [
237        1.0 / (1.0 - x) for x in [obs_cdf_intpol, mod_cdf_intpol, sce_cdf]
238    ]
239
240    adapted_cdf = 1 - 1.0 / (obs_inverse * sce_inverse / mod_inverse)
241    adapted_cdf[adapted_cdf < 0.0] = 0.0
242
243    initial = (
244        gamma.ppf(np.sort(adapted_cdf), *obs_gamma)
245        * gamma.ppf(sce_cdf, *sce_gamma)
246        / gamma.ppf(sce_cdf, *mod_gamma)
247    )
248
249    mod_frequency = 1.0 * mod_raindays.shape[0] / mod_data.shape[0]
250    sce_frequency = 1.0 * sce_raindays.shape[0] / sce_data.shape[0]
251
252    days_min = len(sce_raindays) * sce_frequency / mod_frequency
253
254    expected_sce_raindays = int(min(days_min, len(sce_data)))
255
256    sce_argsort = np.argsort(sce_data)
257    correction = np.zeros(len(sce_data))
258
259    if len(sce_raindays) > expected_sce_raindays:
260        initial = np.interp(
261            np.linspace(1, len(sce_raindays), expected_sce_raindays),
262            np.linspace(1, len(sce_raindays), len(sce_raindays)),
263            initial,
264        )
265    else:
266        initial = np.hstack(
267            (np.zeros(expected_sce_raindays - len(sce_raindays)), initial)
268        )
269
270    if expected_sce_raindays > 0:
271        correction[sce_argsort[-expected_sce_raindays:]] = initial
272    # correction = pd.Series(correction, index=sce_data.index)
273    return correction

276def normal_correction(obs_data, mod_data, sce_data, cdf_threshold=0.9999999):
277    """
278    Correct scenario data with normal scaled distribution mapping.
279
280    Each input is detrended before fitting a normal distribution. The corrected
281    values are then shifted so the scenario trend is added back.
282
283    Parameters
284    ----------
285    obs_data : array-like
286        Reference or observed data.
287    mod_data : array-like
288        Model data over the reference period.
289    sce_data : array-like
290        Scenario/model data to correct.
291    cdf_threshold : float, default 0.9999999
292        Lower and upper CDF clipping threshold used to avoid infinite tails.
293
294    Returns
295    -------
296    numpy.ndarray
297        Corrected scenario data with the same length as `sce_data`.
298
299    Examples
300    --------
301    ```python
302    corrected = normal_correction(reference, model, scenario)
303    ```
304    """
305    obs_len, mod_len, sce_len = [len(x) for x in [obs_data, mod_data, sce_data]]
306    obs_mean, mod_mean, sce_mean = [x.mean() for x in [obs_data, mod_data, sce_data]]
307    obs_detrended, mod_detrended, sce_detrended = [
308        detrend(x) for x in [obs_data, mod_data, sce_data]
309    ]
310    obs_norm, mod_norm, sce_norm = [
311        norm.fit(x) for x in [obs_detrended, mod_detrended, sce_detrended]
312    ]
313
314    obs_cdf = norm.cdf(np.sort(obs_detrended), *obs_norm)
315    mod_cdf = norm.cdf(np.sort(mod_detrended), *mod_norm)
316    sce_cdf = norm.cdf(np.sort(sce_detrended), *sce_norm)
317
318    obs_cdf = np.maximum(np.minimum(obs_cdf, cdf_threshold), 1 - cdf_threshold)
319    mod_cdf = np.maximum(np.minimum(mod_cdf, cdf_threshold), 1 - cdf_threshold)
320    sce_cdf = np.maximum(np.minimum(sce_cdf, cdf_threshold), 1 - cdf_threshold)
321
322    sce_diff = sce_data - sce_detrended
323    sce_argsort = np.argsort(sce_detrended)
324
325    obs_cdf_intpol = np.interp(
326        np.linspace(1, obs_len, sce_len), np.linspace(1, obs_len, obs_len), obs_cdf
327    )
328    mod_cdf_intpol = np.interp(
329        np.linspace(1, mod_len, sce_len), np.linspace(1, mod_len, mod_len), mod_cdf
330    )
331    obs_cdf_shift, mod_cdf_shift, sce_cdf_shift = [
332        (x - 0.5) for x in [obs_cdf_intpol, mod_cdf_intpol, sce_cdf]
333    ]
334
335    obs_inverse, mod_inverse, sce_inverse = [
336        1.0 / (0.5 - np.abs(x)) for x in [obs_cdf_shift, mod_cdf_shift, sce_cdf_shift]
337    ]
338
339    adapted_cdf = np.sign(obs_cdf_shift) * (
340        1.0 - 1.0 / (obs_inverse * sce_inverse / mod_inverse)
341    )
342    adapted_cdf[adapted_cdf < 0] += 1.0
343    adapted_cdf = np.maximum(np.minimum(adapted_cdf, cdf_threshold), 1 - cdf_threshold)
344
345    xvals = norm.ppf(np.sort(adapted_cdf), *obs_norm) + obs_norm[-1] / mod_norm[-1] * (
346        norm.ppf(sce_cdf, *sce_norm) - norm.ppf(sce_cdf, *mod_norm)
347    )
348
349    xvals -= xvals.mean()
350    xvals += obs_mean + (sce_mean - mod_mean)
351
352    correction = np.zeros(sce_len)
353    correction[sce_argsort] = xvals
354    correction += sce_diff - sce_mean
355    # correction = pd.Series(correction, index=sce_data.index)
356    return correction
