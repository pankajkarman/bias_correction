bias_correction

Bias-correction methods for NumPy/Pandas series and Xarray objects.

The module adjusts scenario/model data against a trusted reference data set. Xarray inputs can be multi-dimensional, such as (time, lat, lon) climate grids, as long as they include the time dimension used for correction. Implemented methods include:

  • basic_quantile: empirical quantile mapping.
  • modified_quantile: modified quantile mapping with median and IQR scaling.
  • gamma_mapping: scaled distribution mapping for non-negative, precipitation-like data.
  • normal_mapping: scaled distribution mapping for approximately normally distributed data.

Installation

pip install bias-correction

Install the latest code directly from GitHub:

pip install git+https://github.com/pankajkarman/bias_correction.git

Usage

Use BiasCorrection for one-dimensional NumPy arrays or Pandas Series:

from bias_correction import BiasCorrection

bc = BiasCorrection(reference, model, scenario)
corrected = bc.correct(method="modified_quantile")

Use XBiasCorrection for Xarray DataArray or Dataset inputs:

from bias_correction import XBiasCorrection

xbc = XBiasCorrection(reference, model, scenario, dim="time")
corrected = xbc.correct(method="gamma_mapping", lower_limit=0.1)

For multi-dimensional Xarray data, pass the name of the time dimension with dim. Corrections are applied along that dimension at each remaining grid point:

reference = xr.DataArray(ref_values, dims=["time", "lat", "lon"])
model = xr.DataArray(model_values, dims=["time", "lat", "lon"])
scenario = xr.DataArray(scenario_values, dims=["time", "lat", "lon"])

xbc = XBiasCorrection(reference, model, scenario, dim="time")
corrected = xbc.correct(method="modified_quantile")

reference is the observed or trusted data, model is the model data over the reference period, and scenario is the model-period data to correct, such as a future projection or forecast.

119def _validate_1d_inputs(*arrays):
120    """Raise a clear error when the NumPy/Pandas API receives gridded input."""
121    for name, array in arrays:
122        if np.asarray(array).ndim != 1:
123            raise ValueError(
124                f"{name} must be one-dimensional. "
125                "Use XBiasCorrection for gridded or multidimensional data."
126            )

129def _correction_skipna(
130    correction_func,
131    obs_data,
132    mod_data,
133    sce_data,
134    min_count=1,
135    **kwargs,
136):
137    """Run a correction on finite values and preserve NaNs in scenario output."""
138    obs_data = np.asarray(obs_data, dtype=float)
139    mod_data = np.asarray(mod_data, dtype=float)
140    sce_data = np.asarray(sce_data, dtype=float)
141
142    corrected = np.full(sce_data.shape, np.nan)
143    obs_data = obs_data[np.isfinite(obs_data)]
144    mod_data = mod_data[np.isfinite(mod_data)]
145    sce_mask = np.isfinite(sce_data)
146    finite_sce_data = sce_data[sce_mask]
147
148    if (
149        len(obs_data) < min_count
150        or len(mod_data) < min_count
151        or len(finite_sce_data) < min_count
152    ):
153        return corrected
154
155    corrected[sce_mask] = correction_func(
156        obs_data, mod_data, finite_sce_data, **kwargs
157    )
158    return corrected

161def _gamma_correction_skipna(
162    obs_data, mod_data, sce_data, lower_limit=0.1, cdf_threshold=0.9999999
163):
164    """Run gamma correction on finite values, tolerating all-dry scenario cells."""
165    obs_data = np.asarray(obs_data, dtype=float)
166    mod_data = np.asarray(mod_data, dtype=float)
167    sce_data = np.asarray(sce_data, dtype=float)
168
169    corrected = np.full(sce_data.shape, np.nan)
170    obs_data = obs_data[np.isfinite(obs_data)]
171    mod_data = mod_data[np.isfinite(mod_data)]
172    sce_mask = np.isfinite(sce_data)
173    finite_sce_data = sce_data[sce_mask]
174
175    if len(obs_data) == 0 or len(mod_data) == 0 or len(finite_sce_data) == 0:
176        return corrected
177
178    if not np.any(finite_sce_data >= lower_limit):
179        corrected[sce_mask] = 0.0
180        return corrected
181
182    if not np.any(obs_data >= lower_limit) or not np.any(mod_data >= lower_limit):
183        return corrected
184
185    corrected[sce_mask] = gamma_correction(
186        obs_data,
187        mod_data,
188        finite_sce_data,
189        lower_limit=lower_limit,
190        cdf_threshold=cdf_threshold,
191    )
192    return corrected

195def quantile_correction(obs_data, mod_data, sce_data, modified=True, skipna=False):
196    """
197    Correct scenario data with empirical quantile mapping.
198
199    Parameters
200    ----------
201    obs_data : array-like
202        Reference or observed data.
203    mod_data : array-like
204        Model data over the reference period.
205    sce_data : array-like
206        Scenario/model data to correct.
207    modified : bool, default True
208        If True, apply the modified quantile correction using the median and
209        interquartile range adjustment. If False, apply basic quantile mapping.
210    skipna : bool, default False
211        If True, ignore non-finite values in `obs_data` and `mod_data`, correct
212        finite `sce_data` values, and preserve non-finite scenario values.
213
214    Returns
215    -------
216    array-like
217        Corrected scenario data with the same shape as `sce_data`.
218
219    Examples
220    --------
221    Basic quantile mapping:
222
223    ```python
224    corrected = quantile_correction(reference, model, scenario, modified=False)
225    ```
226
227    Modified quantile mapping:
228
229    ```python
230    corrected = quantile_correction(reference, model, scenario, modified=True)
231    ```
232    """
233    if skipna:
234        return _correction_skipna(
235            quantile_correction,
236            obs_data,
237            mod_data,
238            sce_data,
239            modified=modified,
240            skipna=False,
241        )
242
243    cdf = ECDF(mod_data, side="left")
244    p = cdf(sce_data) * 100
245    cor = np.subtract(*[np.nanpercentile(x, p) for x in [obs_data, mod_data]])
246    if modified:
247        mid = np.subtract(*[np.nanpercentile(x, 50) for x in [obs_data, mod_data]])
248        g = np.true_divide(*[np.nanpercentile(x, 50) for x in [obs_data, mod_data]])
249
250        iqr_obs_data = np.subtract(*np.nanpercentile(obs_data, [75, 25]))
251        iqr_mod_data = np.subtract(*np.nanpercentile(mod_data, [75, 25]))
252
253        f = np.true_divide(iqr_obs_data, iqr_mod_data)
254        cor = g * mid + f * (cor - mid)
255        return sce_data + cor
256    else:
257        return sce_data + cor

260def gamma_correction(
261    obs_data,
262    mod_data,
263    sce_data,
264    lower_limit=0.1,
265    cdf_threshold=0.9999999,
266    skipna=False,
267):
268    """
269    Correct precipitation-like scenario data with gamma distribution mapping.
270
271    Values below `lower_limit` are treated as dry or zero events. Corrected
272    wet-day values are assigned back to the largest scenario values so dry-day
273    frequency is preserved according to the model/reference frequency ratio.
274
275    Parameters
276    ----------
277    obs_data : array-like
278        Reference or observed data.
279    mod_data : array-like
280        Model data over the reference period.
281    sce_data : array-like
282        Scenario/model data to correct.
283    lower_limit : float, default 0.1
284        Minimum value considered a wet day.
285    cdf_threshold : float, default 0.9999999
286        Upper CDF bound used to avoid infinite distribution tails.
287    skipna : bool, default False
288        If True, ignore non-finite values in `obs_data` and `mod_data`, correct
289        finite `sce_data` values, and preserve non-finite scenario values.
290
291    Returns
292    -------
293    numpy.ndarray
294        Corrected scenario data with the same length as `sce_data`.
295
296    Examples
297    --------
298    ```python
299    corrected = gamma_correction(
300        reference,
301        model,
302        scenario,
303        lower_limit=0.1,
304    )
305    ```
306    """
307    if skipna:
308        return _gamma_correction_skipna(
309            obs_data,
310            mod_data,
311            sce_data,
312            lower_limit=lower_limit,
313            cdf_threshold=cdf_threshold,
314        )
315
316    obs_raindays, mod_raindays, sce_raindays = [
317        x[x >= lower_limit] for x in [obs_data, mod_data, sce_data]
318    ]
319    obs_gamma, mod_gamma, sce_gamma = [
320        gamma.fit(x) for x in [obs_raindays, mod_raindays, sce_raindays]
321    ]
322
323    obs_cdf = gamma.cdf(np.sort(obs_raindays), *obs_gamma)
324    mod_cdf = gamma.cdf(np.sort(mod_raindays), *mod_gamma)
325    sce_cdf = gamma.cdf(np.sort(sce_raindays), *sce_gamma)
326
327    obs_cdf[obs_cdf > cdf_threshold] = cdf_threshold
328    mod_cdf[mod_cdf > cdf_threshold] = cdf_threshold
329    sce_cdf[sce_cdf > cdf_threshold] = cdf_threshold
330
331    obs_cdf_intpol = np.interp(
332        np.linspace(1, len(obs_raindays), len(sce_raindays)),
333        np.linspace(1, len(obs_raindays), len(obs_raindays)),
334        obs_cdf,
335    )
336
337    mod_cdf_intpol = np.interp(
338        np.linspace(1, len(mod_raindays), len(sce_raindays)),
339        np.linspace(1, len(mod_raindays), len(mod_raindays)),
340        mod_cdf,
341    )
342
343    obs_inverse, mod_inverse, sce_inverse = [
344        1.0 / (1.0 - x) for x in [obs_cdf_intpol, mod_cdf_intpol, sce_cdf]
345    ]
346
347    adapted_cdf = 1 - 1.0 / (obs_inverse * sce_inverse / mod_inverse)
348    adapted_cdf[adapted_cdf < 0.0] = 0.0
349
350    initial = (
351        gamma.ppf(np.sort(adapted_cdf), *obs_gamma)
352        * gamma.ppf(sce_cdf, *sce_gamma)
353        / gamma.ppf(sce_cdf, *mod_gamma)
354    )
355
356    mod_frequency = 1.0 * mod_raindays.shape[0] / mod_data.shape[0]
357    sce_frequency = 1.0 * sce_raindays.shape[0] / sce_data.shape[0]
358
359    days_min = len(sce_raindays) * sce_frequency / mod_frequency
360
361    expected_sce_raindays = int(min(days_min, len(sce_data)))
362
363    sce_argsort = np.argsort(sce_data)
364    correction = np.zeros(len(sce_data))
365
366    if len(sce_raindays) > expected_sce_raindays:
367        initial = np.interp(
368            np.linspace(1, len(sce_raindays), expected_sce_raindays),
369            np.linspace(1, len(sce_raindays), len(sce_raindays)),
370            initial,
371        )
372    else:
373        initial = np.hstack(
374            (np.zeros(expected_sce_raindays - len(sce_raindays)), initial)
375        )
376
377    if expected_sce_raindays > 0:
378        correction[sce_argsort[-expected_sce_raindays:]] = initial
379    # correction = pd.Series(correction, index=sce_data.index)
380    return correction

383def normal_correction(
384    obs_data, mod_data, sce_data, cdf_threshold=0.9999999, skipna=False
385):
386    """
387    Correct scenario data with normal scaled distribution mapping.
388
389    Each input is detrended before fitting a normal distribution. The corrected
390    values are then shifted so the scenario trend is added back.
391
392    Parameters
393    ----------
394    obs_data : array-like
395        Reference or observed data.
396    mod_data : array-like
397        Model data over the reference period.
398    sce_data : array-like
399        Scenario/model data to correct.
400    cdf_threshold : float, default 0.9999999
401        Lower and upper CDF clipping threshold used to avoid infinite tails.
402    skipna : bool, default False
403        If True, ignore non-finite values in `obs_data` and `mod_data`, correct
404        finite `sce_data` values, and preserve non-finite scenario values.
405
406    Returns
407    -------
408    numpy.ndarray
409        Corrected scenario data with the same length as `sce_data`.
410
411    Examples
412    --------
413    ```python
414    corrected = normal_correction(reference, model, scenario)
415    ```
416    """
417    if skipna:
418        return _correction_skipna(
419            normal_correction,
420            obs_data,
421            mod_data,
422            sce_data,
423            min_count=2,
424            cdf_threshold=cdf_threshold,
425            skipna=False,
426        )
427
428    obs_len, mod_len, sce_len = [len(x) for x in [obs_data, mod_data, sce_data]]
429    obs_mean, mod_mean, sce_mean = [x.mean() for x in [obs_data, mod_data, sce_data]]
430    obs_detrended, mod_detrended, sce_detrended = [
431        detrend(x) for x in [obs_data, mod_data, sce_data]
432    ]
433    obs_norm, mod_norm, sce_norm = [
434        norm.fit(x) for x in [obs_detrended, mod_detrended, sce_detrended]
435    ]
436
437    obs_cdf = norm.cdf(np.sort(obs_detrended), *obs_norm)
438    mod_cdf = norm.cdf(np.sort(mod_detrended), *mod_norm)
439    sce_cdf = norm.cdf(np.sort(sce_detrended), *sce_norm)
440
441    obs_cdf = np.maximum(np.minimum(obs_cdf, cdf_threshold), 1 - cdf_threshold)
442    mod_cdf = np.maximum(np.minimum(mod_cdf, cdf_threshold), 1 - cdf_threshold)
443    sce_cdf = np.maximum(np.minimum(sce_cdf, cdf_threshold), 1 - cdf_threshold)
444
445    sce_diff = sce_data - sce_detrended
446    sce_argsort = np.argsort(sce_detrended)
447
448    obs_cdf_intpol = np.interp(
449        np.linspace(1, obs_len, sce_len), np.linspace(1, obs_len, obs_len), obs_cdf
450    )
451    mod_cdf_intpol = np.interp(
452        np.linspace(1, mod_len, sce_len), np.linspace(1, mod_len, mod_len), mod_cdf
453    )
454    obs_cdf_shift, mod_cdf_shift, sce_cdf_shift = [
455        (x - 0.5) for x in [obs_cdf_intpol, mod_cdf_intpol, sce_cdf]
456    ]
457
458    obs_inverse, mod_inverse, sce_inverse = [
459        1.0 / (0.5 - np.abs(x)) for x in [obs_cdf_shift, mod_cdf_shift, sce_cdf_shift]
460    ]
461
462    adapted_cdf = np.sign(obs_cdf_shift) * (
463        1.0 - 1.0 / (obs_inverse * sce_inverse / mod_inverse)
464    )
465    adapted_cdf[adapted_cdf < 0] += 1.0
466    adapted_cdf = np.maximum(np.minimum(adapted_cdf, cdf_threshold), 1 - cdf_threshold)
467
468    xvals = norm.ppf(np.sort(adapted_cdf), *obs_norm) + obs_norm[-1] / mod_norm[-1] * (
469        norm.ppf(sce_cdf, *sce_norm) - norm.ppf(sce_cdf, *mod_norm)
470    )
471
472    xvals -= xvals.mean()
473    xvals += obs_mean + (sce_mean - mod_mean)
474
475    correction = np.zeros(sce_len)
476    correction[sce_argsort] = xvals
477    correction += sce_diff - sce_mean
478    # correction = pd.Series(correction, index=sce_data.index)
479    return correction

482class BiasCorrection(object):
483    """
484    Bias-correction wrapper for one-dimensional NumPy arrays and Pandas Series.
485
486    Parameters
487    ----------
488    obs_data : array-like
489        Reference or observed data.
490    mod_data : array-like
491        Model data over the reference period.
492    sce_data : array-like
493        Scenario/model data to correct.
494
495    Notes
496    -----
497    If `sce_data` is a Pandas `Series`, corrected output is returned as a
498    `Series` with the original index preserved. Use `XBiasCorrection` for
499    gridded or other multidimensional data.
500    """
501
502    def __init__(self, obs_data, mod_data, sce_data):
503        _validate_1d_inputs(
504            ("obs_data", obs_data), ("mod_data", mod_data), ("sce_data", sce_data)
505        )
506        self.obs_data = obs_data
507        self.mod_data = mod_data
508        self.sce_data = sce_data
509
510    def correct(
511        self,
512        method="modified_quantile",
513        lower_limit=0.1,
514        cdf_threshold=0.9999999,
515        skipna=False,
516    ):
517        """
518        Apply a bias-correction method.
519
520        Parameters
521        ----------
522        method : {"modified_quantile", "basic_quantile", "gamma_mapping", "normal_mapping"}
523            Correction method to apply.
524        lower_limit : float, default 0.1
525            Wet-day threshold used by `gamma_mapping`.
526        cdf_threshold : float, default 0.9999999
527            CDF clipping threshold used by distribution mapping methods.
528        skipna : bool, default False
529            If True, ignore non-finite values in `obs_data` and `mod_data`,
530            correct finite `sce_data` values, and preserve non-finite scenario
531            values.
532
533        Returns
534        -------
535        array-like or pandas.Series
536            Corrected scenario data.
537
538        Examples
539        --------
540        ```python
541        corrected = bc.correct(method="basic_quantile")
542        corrected = bc.correct(method="modified_quantile")
543        corrected = bc.correct(method="gamma_mapping", lower_limit=0.1)
544        corrected = bc.correct(method="normal_mapping")
545        ```
546        """
547        if method == "gamma_mapping":
548            corrected = gamma_correction(
549                self.obs_data,
550                self.mod_data,
551                self.sce_data,
552                lower_limit=lower_limit,

316

318        x[x >= lower_limit] for x in [obs_data, mod_data, sce_data]

321        gamma.fit(x) for x in [obs_raindays, mod_raindays, sce_raindays]

326    sce_cdf = gamma.cdf(np.sort(sce_raindays), *sce_gamma)
327
328    obs_cdf[obs_cdf > cdf_threshold] = cdf_threshold
329    mod_cdf[mod_cdf > cdf_threshold] = cdf_threshold
330    sce_cdf[sce_cdf > cdf_threshold] = cdf_threshold
331
332    obs_cdf_intpol = np.interp(
333        np.linspace(1, len(obs_raindays), len(sce_raindays)),
334        np.linspace(1, len(obs_raindays), len(obs_raindays)),
335        obs_cdf,
336    )
337
338    mod_cdf_intpol = np.interp(
339        np.linspace(1, len(mod_raindays), len(sce_raindays)),

346    ]

349    adapted_cdf[adapted_cdf < 0.0] = 0.0
350
351    initial = (
352        gamma.ppf(np.sort(adapted_cdf), *obs_gamma)

361

363

372        )

409    numpy.ndarray
